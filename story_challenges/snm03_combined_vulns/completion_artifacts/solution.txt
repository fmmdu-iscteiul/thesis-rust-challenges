/// Snemeis Challenge 3 fixed version:
/// Validate every parsing step explicitly and reject malformed or unexpected input.

pub struct EscalationRequest {
    pub ticket_id: u32,
    pub level: u8,
    pub team: String,
}

pub fn parse_escalation(input: &str) -> Result<EscalationRequest, String> {
    let mut ticket_id = None;
    let mut level = None;
    let mut team = None;

    for part in input.split(';') {
        let (key, value) = part
            .split_once('=')
            .ok_or_else(|| format!("malformed escalation segment: {:?}", part))?;

        match key.trim() {
            "ticket" => {
                let parsed = value
                    .trim()
                    .parse::<u32>()
                    .map_err(|_| "invalid ticket value".to_string())?;
                ticket_id = Some(parsed);
            }
            "level" => {
                let parsed = value
                    .trim()
                    .parse::<u8>()
                    .map_err(|_| "invalid level value".to_string())?;
                level = Some(parsed);
            }
            "team" => {
                let parsed = value.trim();
                if parsed.is_empty() {
                    return Err("missing team value".to_string());
                }
                team = Some(parsed.to_string());
            }
            other => {
                return Err(format!("unknown escalation field: {}", other));
            }
        }
    }

    Ok(EscalationRequest {
        ticket_id: ticket_id.ok_or_else(|| "missing required ticket field".to_string())?,
        level: level.ok_or_else(|| "missing required level field".to_string())?,
        team: team.ok_or_else(|| "missing required team field".to_string())?,
    })
}

pub fn process_escalation(req: &EscalationRequest) -> Result<String, String> {
    if !(1..=3).contains(&req.level) {
        return Err(format!(
            "escalation level {} is out of range (allowed: 1-3)",
            req.level
        ));
    }

    Ok(format!(
        "Ticket {} escalated to level {} -> routed to team {}",
        req.ticket_id, req.level, req.team
    ))
}

pub fn handle_escalation(input: &str) -> Result<String, String> {
    let req = parse_escalation(input)?;
    process_escalation(&req)
}
